// ⚡ เปลี่ยน URL ตรงนี้ให้เป็นลิงก์ Web App ที่ได้จาก Google Sheets ทดลองของคุณ const WEB_APP_URL = "https://google.com"; const SHEETS_SECRET = "123456"; // รหัสร้านค้า // ========================================== // ⚡ ฟังก์ชันโหลดข้อมูลพนักงานความเร็วสูง // ========================================== function loadEmployeesFast() { const selectElement = document.getElementById('employee-select'); // เปลี่ยน ID ให้ตรงกับหน้าเว็บคุณ if (!selectElement) return; // ส่งคำขอแบบม้วนเดียวจบ fetch(WEB_APP_URL, { method: 'POST', body: JSON.stringify({ action: 'list', prefix: 'emp_', secret: SHEETS_SECRET }) }) .then(res => res.json()) .then(res => { if (res.error) { console.error('Error:', res.error); return; } selectElement.innerHTML = ''; const allEmpData = res.data; // ก้อนข้อมูลพนักงานทั้งหมดที่แถมมา // ดึงข้อมูลจากหน่วยความจำเครื่องมาใส่ในช่องตัวเลือกทันที (ไม่ต้องยิงขอข้อมูลทีละคน) res.keys.forEach(key => { const empName = allEmpData[key]; if (empName) { const option = document.createElement('option'); option.value = key; option.textContent = empName; selectElement.appendChild(option); } }); }) .catch(err => console.error('Fetch error:', err)); } // ========================================== // ⚡ ฟังก์ชันโหลดหน้าข้อมูลเงินเชื่อความเร็วสูง // ========================================== function loadCreditDataFast() { const creditContainer = document.getElementById('credit-list-container'); // เปลี่ยน ID ให้ตรงกับหน้าตารางเงินเชื่อคุณ if (!creditContainer) return; fetch(WEB_APP_URL, { method: 'POST', body: JSON.stringify({ action: 'list', prefix: 'credit_', secret: SHEETS_SECRET }) }) .then(res => res.json()) .then(res => { if (res.error) { console.error('Error:', res.error); return; } creditContainer.innerHTML = ''; // ล้างตารางเก่า const allCreditData = res.data; // ก้อนข้อมูลประวัติเงินเชื่อทั้งหมดที่แถมมา res.keys.forEach(key => { const creditDetail = allCreditData[key]; if (creditDetail) { // 解析ข้อมูลเงินเชื่อ (หากคุณเซฟเป็นแบบ JSON หรือ Text) // ตัวอย่างการวาดแถวตารางเงินเชื่อในเครื่องทันทีโดยไม่ช้าค้าง const row = document.createElement('div'); row.className = 'line-item'; row.innerHTML = ` `; creditContainer.appendChild(row); } }); }) .catch(err => console.error('Fetch error:', err)); } // เรียกใช้งานฟังก์ชันโหลดด่วนตอนเปิดหน้าแอปพลิเคชัน window.addEventListener('DOMContentLoaded', () => { loadEmployeesFast(); loadCreditDataFast(); });
